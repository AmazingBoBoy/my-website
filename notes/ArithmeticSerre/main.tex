\section{Chapter 1: Finite Fields and Quadratic Reciprocity}
This is a list of proofs of nontrivial results in Chapter 1 of \textit{A course in Arithmetic} by J.P. Serre.
\\

Let $K$ be a finite field with $q$ elements. 
\begin{theorem}
    Let $u$ be an integer $\geq 0$. The sum 
    \[S(X^u)=\sum_{x\in K} x^u=\begin{cases}
        -1 & \text{if } u>0 \text{ and } (q-1)\mid u\\
        0 & \text{otherwise}
    \end{cases}\]
$u=0$ is subtle here. We define $x^0=1$ for all $x\in K$ (this definition is nice when we work with polynomials because we can write $1$ as $x^0$ to maintain symmetry), so $S(X^0)=q=0$.
\end{theorem}
The proof is easy. Use primitive roots. 
\\
The variant proof: $x\mapsto x^u$ is a group homomorphism from $K^\times $ to itself. The kernel is all $x$ such that $x$ is a root of $x^{\gcd(u,q-1)}-1$. So the image has size $\frac{q-1}{\gcd(u,q-1)}$, which implies the image is just the $d=\frac{q-1}{\gcd(u,q-1)}$ roots of unity. So the sum is $0$ unless $d=1$ (sum of d-th roots of unity is $0$ if $d\geq 2$ s coprime to $p$).
\begin{theorem}[Chevalley-Warning Theorem]
    Let $P_1,\ldots,P_r$ be polynomials in $n$ variables over a finite field $K$ of characteristic $p$. If the sum of their degrees is less than $n$, then the number of common solutions to the equations $P_1=\cdots=P_r=0$ is divisible by $p$. 
\end{theorem}
\begin{proof}
This uses a common trick in counting roots over finite fields. Consider the polynomial
\[Q=\prod_{i=1}^r (1-P_i^{q-1})\]
The number of common solutions is equal to $\sum_{x\in K^n} Q(x)$ since if $P_i(x)=0$ for all $i$, then $Q(x)=1$, otherwise $Q(x)=0$. 
\\

Expand this. For each monomial in the expansion $x^{u_1}_1\cdots x^{u_n}_n$, since $\sum \deg(P_i)(q-1)<n(q-1)$, we have $u_j$ is not divisible less than $q-1$ for some $j$. So by the previous theorem, the sum over $K^n$ of this monomial is $0$. (that's why we define $x^0=1$ above). So the number of common solutions is $0$ mod $p$.
\end{proof}
\begin{corollary}
    All quadratic forms in at least 3 variables over a finite field have nontrivial zeros. 
\end{corollary}
\begin{theorem}
    $\left( \frac{2}{p}\right)=(-1)^{\frac{p^2-1}{8}}$
\end{theorem}
\begin{proof}
    Let $\alpha$ be the primitve 8th root of unity $\alpha$ in $\ol{\FF_p}$. Then, $y^2=(\alpha+\alpha^{-1})^2=2$. If $p\equiv \pm 1\mod 8$, then $y^p=y$, so $2$ is a square in $\FF_p$. If $p\equiv \pm 3\mod 8$, then $y^p=-y$, so $2$ is not a square in $\FF_p$.
\end{proof}
\begin{theorem}[Law of Quadratic Reciprocity]
    For two odd primes $p$ and $q$, we have
    \[\left( \frac{p}{q}\right)\left( \frac{q}{p}\right)=(-1)^{\frac{(p-1)(q-1)}{4}}\]
    
\end{theorem}
The key is to consider the Gauss sum: Let $\zeta$ be a primitive $p$-th root of unity in $\ol{\FF_p}$, then define
\[y=\sum_{x\in \FF_p}\left( \frac{x}{p}\right) \zeta^x\] 
Then we have $y^2=\left( \frac{-1}{p}\right)p=(-1)^{\frac{p-1}{2}}p$ and $y^{q-1}=\left( \frac{q}{p}\right)$. Combining these two gives the result.

We can in general define a norm on a number field $K$ as follows: 
\begin{definition}
    $\norm{\cdot}: K\to \RR_{\geq 0}$ is a norm if
    \begin{enumerate}
        \item $\norm{x}=0$ iff $x=0$
        \item $\norm{xy}=\norm{x}\norm{y}$
        \item $\norm{x+y}\leq \norm{x}+\norm{y}$
    \end{enumerate}
\end{definition}
\begin{corollary}
    $\norm{1}=1$ and $\norm{-1}=1$
\end{corollary}
\begin{definition}
    Norm $N_1$ and $N_2$ are equivalent if there exists a real number $c$ such that $N_2(x)=N_1(x)^a$ for all $x$.
\end{definition}
\begin{theorem}[Ostrowski's Theorem]
    Any non-trivial norm on $\QQ$ is equivalent to either the usual absolute value $\norm{\cdot}_\infty$ or to $\norm{\cdot}_p$ for some prime $p$.
\end{theorem}
\begin{proof}
    We consider two cases:
    \\

    \textbf{Case 1:} Assumn $\norm{n}\leq 1$ for all $n\in \ZZ$. If there exists $n$ such that $\norm{n}<1$. Then there must be a prime $p$ dividing $n$ such that $\norm{p}<1$. We claim that $p$ is the only prime such that $\norm{p}<1$. Let $q$ be another prime such that $\norm{q}<1$. We can find $N$ such that $\norm{p}^N$ and $\norm{q}^N$ are both less than $\frac{1}{2}$. By Bezout's lemma, we can find integers $a,b$ such that $ap^N+bq^N=1$. So $1=\norm{1}=\norm{ap^N+bq^N}\leq \norm{a}\norm{p}^N+\norm{b}\norm{q}^N<\frac{\norm{a}+\norm{b}}{2}\leq 1$, which is a contradiction. So in this case, $\norm{\cdot}$ is equivalent to $\norm{\cdot}_p$.
    \\

    The rest is sort of cumbersome. 
\end{proof}

\end{example}

\begin{theorem}
    If $n\geq 3$, $x^n+y^n=z^n$ has no non-trivial solutions in $\CC[x]$. 
\end{theorem}
\begin{proof}
    Assume $f,g h$ are non-constant polynomials such that $f^n+g^n=h^n$. We further assume they $\max{\deg(f), \deg(g), \deg(h)}$ is minimal among all such triples. Therefore, $\gcd(f,g)=\gcd(g,h)=\gcd(h,f)=1$. (otherwise, we can divide the common factor to get a smaller degree triple).
    \\

    $f^n=h^n-g^n=\prod (h-\zeta^i g)$ where $\zeta$ is a primitive $n$-th root of unity. Notice that $\gcd(h-\zeta^i g,h-\zeta^j g)=\gcd(g,h)$ for $i\neq j$. So all $h-\zeta^i g$ are coprime to each other. Thus, $h-g=a^n$, $h-\zeta g=b^n$, $z-\zeta^2 g=c^n$ for some non-constant polynomials $a,b,c\in \CC[x]$. We can easily find $p, q,r \in \CC$ such that $(pa)^n+(qb)^n+(rc)^n=0$. This contradicts the minimality of our choice of $f,g,h$.
\end{proof}

\section{Chapter 2: $p$-adic numbers}
\subsection{$\ZZ_p$ and $\QQ_p$}
Let $A_n=\ZZ/p^n\ZZ$. There is a natural projection map $\phi_n: A_{n}\to A_{n-1}$.
\begin{definition}
    We define $\ZZ_p$ as the limit (in category of rings) of 
    \[\cdots \taking{\phi_{n+1}} A_n \taking{\phi_{n}} A_{n-1} \taking{\phi_{n-1}} \cdots \taking{\phi_{2}} A_1\]
\end{definition}
By the definition, any $x\in \ZZ_p$ can be represented as a seqeunce $(\ldots,x_2,x_1)$ where $x_n\in A_n$ and $\phi_n(x_n)=x_{n-1}$. Alternatively, we can represent $x$ as an infinite sum
\[x=a_0+a_1 p + a_2 p^2 + \cdots\]
where $a_i\in \{0,1,\ldots,p-1\}$. 

We can also define the topology on $\ZZ_p$ as the induced topology from the product topology on $\prod A_n$ where $A_n$ is given the discrete topology. 
\begin{theorem}
    $\ZZ_p$ is compact.
\end{theorem}
\begin{proof}
    We will need a lemma from topology. 
    \begin{lemma}
        \textbf{Closed subset of a compact space is compact.}
    \begin{subproof}
            If $K$ is compact and $F\subseteq K$ is closed, then for any open covering of $F$, we can add $K\setminus F$ to get an open covering of $K$. So there is a finite subcovering, removing $K\setminus F$ gives a finite subcovering of $F$.
        \end{subproof}
    \end{lemma}
    Since $\prod A_n$ is compact by Tychonoff's theorem, it suffices to show $\ZZ_p$ is closed in $\prod A_n$. It suffices show that its complement is open. For any $y\notin \ZZ_p$, there exists some $j$ such that $\phi_{j}(y_{j})\neq y_{j-1}$. There exists an open set $U=\prod U_n$ where $U_n=A_n$ for $n\neq j,j-1$, $U_j=\{y_j\}$ and $U_{j-1}=\{y_{j-1}\}$. It's clear that $U$ is an open neighborhood of $y$ and $U\cap \ZZ_p=\emptyset$. So $\ZZ_p$ is closed.
\end{proof}
\begin{theorem}
    \[0\to A_m \taking{\times p^n} A_{m+n} \taking{\phi_{m+n,n}} A_n \to 0\]
    is an exact seqeunce of abelian groups.
\end{theorem}
This is easy to check. The first injective map is moving the the last $m$ digits $\times p^n$ to the left by $n$ places and fill the last $n$ digits with zero, the second map is just truncating the last $n$ digits. We will use this in $\ZZ_p$.
\begin{theorem}
    \[0\to \ZZ_p \taking{\times p^n} \ZZ_p \to A_n \to 0 \]
    is an exact seqeunce of abelian groups.
\end{theorem}
\begin{proof}
    First, we need to show $\ZZ_p \taking{\times p^n} \ZZ_p$ is injective. Notice that if $\phi_{m+n,n}(x_{m+n})=x_n$, then $p^mx_n$ and $p^mx_{m+n}$ are the same in $A_{m+n}$. Therefore, we conclude that multiplying by $p^n$ is just adding $n$ zeros to the end, that is sending $(\ldots,x_2,x_1)$ to $(\ldots,p^nx_2,p^nx_1,0,\ldots,0)$. Let $j$ be the smallest integer such that the $j$-th digit of $x$ is nonzero (not divisble by $p^j$), then the $j+n$-th digit of $\times p^n x$ is also nonzero since it's not divisible by $p^{j+n}$, so the map is injective.
    \\

    The image of $\times p^n$ is just $p^n\ZZ_p$, which is exactly the kernel of the projection map to $A_n$ since the last $n$ digits are all zero.
\end{proof}
\begin{theorem}
    Two propositions:
    \begin{enumerate}
        \item For an element $x\in \ZZ_p$ to be invertible, it is equivalent to say that it's not divisible by $p$, i.e., the first digit $a_0$ in its expansion is nonzero.
        \item Let $U$ be the group of units in $\ZZ_p$. Then we have the decomposition. For any $x\in \ZZ_p$, we can write 
        \[x=p^nu\]
        where $n\geq 0$ and $u\in U$ in a unique way.
    \end{enumerate}
\end{theorem}
For part 1, if $x_1$ is invertible, $a_i$ are corpime to $p$. By Bezout's lemma, there exists $a_i,b_i\in \ZZ$ such that $a_i x_i + b_i p^i=1$. Let $a=(\ldots,a_2,a_1)$, this is clearly the inverse of $x$ and we can check it's indeed in $\ZZ_p$.
\\

Other parts are easy. 

\begin{proposition}
    propositions of $\ZZ_p$ as rings: 
    \begin{enumerate}
        \item As a ring, we can check that $\ZZ_p$ is an integral domain. In fact, it's a local ring with maximal ideal $p\ZZ_p$.
        \item $\ZZ_p$ is discrete valuation ring. Thus, Euclidean domain, PID and UFD.
    \end{enumerate}
\end{proposition} 
\begin{definition}
    We define $v_p(x)=n$ if $x=p^n u$ where $u$ is a unit in $\ZZ_p$.
    \\
    We define $v_p(0)=+\infty$.
\end{definition}
\begin{proposition}
    $v_p(xy)=v_p(x)+v_p(y)$ and $v_p(x+y)\geq \inf\{v_p(x),v_p(y)\}$ with equality if $v_p(x)\neq v_p(y)$.
\end{proposition}
This allows us to define a metric on $\ZZ_p$ as follows:
\begin{definition}
    $d(x,y)=p^{-v(x-y)}$
\end{definition}
A notbale property is that $d(x,y)=d(x-z,y-z)$ for any $x,y,z\in \ZZ_p$. So the metric is translation invariant. In particular, we have $d(x,y)=d(0,y-x)$.
\\

\begin{theorem}
    The metric space $(\ZZ_p,d)$ is homomorphic to the topological space $\ZZ_p$ defined as the subspace of $\prod A_n$ with the product topology.
\end{theorem}
\begin{proof}
    It suffices to show that the identity map from $(\ZZ_p,d)$ to $\ZZ_p$ is continuous in both directions.
    \\
    For any open set $U\subseteq \ZZ_p$, 
    \[U=\ZZ_p\cap \left(\bigcup_{i\in I}(\prod U_{n, i})\right)=\bigcup_{i\in I}\left((\prod U_{n, i}\cap \ZZ_p)\right)\] 
    where $U_{n,i}=A_n$ for all but finitely many $n$.
    \\

    What does $T_i=\prod U_{n, i}\cap \ZZ_p$ looks like? Let $N$ be the largest integer such that $U_{N,i}\neq A_N$. Then, $T_i$ is the set of $x$ such that the first $N$ digits of $x$ are fixed and the rest can be anything possible, that is $x=(\ldots,a_N,\ldots,a_1)$ where $a_1,\ldots,a_N$ are fixed. The preimage of $T_i$ in $(\ZZ_p,d)$ is precisely the open ball centered at $x$ with radius $p^{-N}$, that is $x+p^N\ZZ_p$. 
    \\

    Conversely, what does any open ball in $(\ZZ_p,d)$ look like in $\ZZ_p$? First, the open ball of $0$ is just ideals $p^n\ZZ_p$. For other point, if the radius of the open abll is $p^{-N}$, for some $N\in\ZZ$. $d(x,y)=d(0,x-y)<p^N$ means $v_p(x-y)>N$, which means the first $N$ digits of $x$ and $y$ are the same. So the open balls are the cosets of $p^n\ZZ_p$. So the preimage of this open ball is the set of $y$ such that the first $N$ digits are fixed, which is clearly open in $\ZZ_p$.
\end{proof}
\begin{proposition}
    propositions of $\ZZ_p$ as a metric space:
    \begin{enumerate}
        \item image of $d(0,x)$ is discrete in $\RR$. So $\ZZ_p$ is totally disconnected.
        \item $\ZZ_p$ is complete under this metric. This follows from the fact that $\ZZ_p$ is compact.
        \item $\ZZ$ is dense in $\ZZ_p$. Just find an integer with the same first $n$ digits as the element in $\ZZ_p$ you want to approximate.
    \end{enumerate}
\end{proposition}

\begin{definition}
    We define $\QQ_p$ as the field of fractions of $\ZZ_p$. Equivalently, it's the localization of $\ZZ_p$ at $U={1,p,p^2,\ldots}$. 
    \\
\end{definition}
\begin{definition}
    For any $x\in \QQ_p$, we can write $x=p^n u$ where $n\in \ZZ$ and $u\in U$. We define $v_p(x)=n$ and $v_p(0)=+\infty$.
\end{definition}
\begin{proposition}
    propositions about $\QQ_p$:
    \begin{enumerate}
        \item $x\in \QQ_p$ is in $\ZZ_p$ iff $v_p(x)\geq 0$.
        \item The metric $d(x,y)=p^{-v_p(x-y)}$ makes $\QQ_p$ a topological field.
        \item $\ZZ_p$ is a clopen subset of $\QQ_p$. 
        \item $\QQ_p$ is locally compact, that is for any $x\in \QQ_p$, there exists open set $U$ and compact set $K$ such that $x\in U\subseteq K$.
        \item $\QQ$ is dense in $\QQ_p$.
    \end{enumerate}
\end{proposition}
For any open neighborhood of $x\in \QQ_p$, it looks like $x+p^n\ZZ_p$ for some $n$ (in fact clopen). Notice that translation and scaling are homeomorphisms, so any neighborhood of $x$ is compact. Others are easy to check. 
\begin{proposition}
    The metric $d$ is an ultrametric, that is it satisfies the strong triangle inequality
    \[d(x,z)\leq \max\{d(x,y),d(y,z)\}\]
    for any $x,y,z\in \QQ_p$.
    \\

    As a result, $d(u_m,u_n)\leq \sup_{k\geq n} d(u_{k},u_{k+1})$ for any sequence $\{u_n\}$ in $\QQ_p$ and any $m>n$. Therefore, a sequence $\{u_n\}$ is Cauchy iff $d(u_n,u_{n+1})\to 0$ as $n\to \infty$ iff $\lim_{n\to \infty} u_{n+1}-u_n=0$.
\end{proposition}
\subsection{$p$-adic equations}
\begin{lemma}
    Let $\cdots \to D_n \to D_{n-1} \to \cdots \to D_1$ be a projective system of non-empty finite sets. Then, the projective limit $\varprojlim D_n$ is non-empty.
\end{lemma}
\begin{proof}
    If each projection map is surjective, then it's clear. So our goal is to turn each map into a surjective map. For each $D_n$, we define $D_{n,p}$ to be the image of $D_{n+p}$ projected to $D_n$. Therefore, for a fixed $n$, we have a decreasing sequence of non-empty finite sets
    \[D_n\supseteq D_{n,1}\supseteq D_{n,2}\supseteq \cdots\]
    Since $D_i$ is finite, there exists $N$ such that $D_{n,N}=D_{n,N+1}=\cdots$. We define $E_n=D_{n,N}$. It's clearly that $E_n$ is non-empty and finite. And $E_n$ is the image of $E_{n+1}$ projected to $D_n$. So we have a projective system of non-empty finite sets with surjective projection maps. Therefore, the projective limit $\varprojlim E_n$ is non-empty. It's clear that $\varprojlim E_n \subseteq \varprojlim D_n$, so $\varprojlim D_n$ is non-empty.
\end{proof}
\begin{definition}
    If $f\in \ZZ_p[x]$ is a polynomial, for $n\geq 1$, we define $f_n\in A_n[x]$ to be the reduction of $f$ mod $p^n$.
\end{definition}
\begin{theorem}
    Let $S$ be a set of polynomials in $\ZZ_p[x_1,\ldots,x_n]$. The following are equivalent:
    \begin{enumerate}
        \item There exists a common root of $S$ in $\ZZ_p^n$. 
        \item For all $m\geq 1$, there exists a common root of $S_m$ (the reduction of $S$ mod $p^m$) in $A_m^n$.
    \end{enumerate}
\end{theorem}
\begin{proof}
    $(1)\implies (2)$ is trivial since we can reduce the common root mod $p^m$ to get a common root in $A_m^n$.
    \\
    
    For $(2)\implies (1)$, for each $m\geq 1$, let $D_m$ be the set of common roots of $S_m$. So we have a project system of non-empty finite sets $\{D_m\}$ with projection maps induced by the natural projection $A_{m}\to A_{m-1}$. By the previous lemma, the projective limit $\varprojlim D_m$ is non-empty. Any element in the projective limit gives a common root of $S$ in $\ZZ_p^n$.
\end{proof}
\begin{definition}
    $x=(x_1, \ldots, x_n)\in (\ZZ_p)^n$ is called primitive if one of $x_i$ is a unit in $\ZZ_p$. One can similarly define primitive elements in $(A_n)^n$.
\end{definition}
\begin{theorem}
    Let $S$ be a set of homogeneous polynomials in $\ZZ_p[x_1,\ldots,x_n]$. The following are equivalent:
    \begin{enumerate}
        \item $S$ has a non trivial common root in $\QQ_p^n$.
        \item There exists a common primitive root of $S$ in $\ZZ_p^n$. 
        \item For all $m\geq 1$, there exists a common primitive root of $S_m$ (the reduction of $S$ mod $p^m$) in $A_m^n$.
    \end{enumerate}
\end{theorem}
This is easy to check. 
\begin{lemma}[Hensel's Lemma]
    Let $f\in \ZZ_p[x]$ be a polynomial. Let $a\in \ZZ_p$, $n, k\in \ZZ$ such that $0\leq 2k <n$, $f(a)\equiv 0\pmod{p^n}$ and $v_p(f'(a))=k$. Then, there exists a  $b\in \ZZ_p$ such that 
    \\
    $f(b)\equiv 0\pmod{p^{n+1}}$, $v_p(f'(b))=k$ and $b\equiv a\pmod{p^{n-k}}$.
\end{lemma}
Our familiar version is the case when $k=0$ and $n=1$. 
\begin{proof}
    Take $b=a+p^{n-k}z$ with $z\in \ZZ_p$ so that $a\equiv b \pmod{p^{n-k}}$. 
    \[f(b)=f(a+p^{n-k}z)=f(a)+p^{n-k}zf'(a)+ p^{2n-2k}z^2t+\cdots\]
Therefore, let $f'(a)=p^kc$ and $f(a)=p^nr$, we have
\[f(b)\equiv p^nr+p^nzc\pmod{p^{n+1}}\]
Note that the second term is suppose to be $(p^{2n-2k}z^2)\frac{f''(a)}{2}$, so where is the $\frac{1}{2}$? This is a real problem if $p=2$ The answer is that when we expand $f(a+x)$, all coefficients of $x^2$ should automatically be in $\ZZ_p$, so $f''(a)$ is divisible by $2$. And here we denote $t=\frac{f''(a)}{2}$
\\

Since $p\nmid c$, we can find $z$ such that $r+zc\equiv 0\pmod{p}$. Lastly, we can easily check that $f'(b)=f'(a+p^{n-k}z)\equiv f'(a)\pmod{p^{n-k}}$, so $v_p(f'(b))=k$.
\end{proof}
\begin{remark}
    Notice that we can do this iteratively to get a solution $y\in \ZZ_p$ such that $f(y)=0$ and $y\equiv a \pmod{p^{n-k}}$. Furthermore, $y$ is unique. 
\end{remark}

\begin{theorem}[Generalization of Hensel's Lemma]
    Let $f\in \ZZ_p[x_1,\ldots,x_m]$ be a polynomial. Let $a=(a_1,\ldots,a_n)\in (\ZZ_p)^m$, $n, k\in \ZZ$  and $1\leq j \leq m$ such that $0\leq 2k <n$, $f(a)\equiv 0\pmod{p^n}$ and $v_p\left(\frac{\partial f}{\partial x_j}(a)\right)=k$. 
    \\

    Then there exists $y\in \ZZ_p^m$ such that $f(y)=0$ and $y\equiv a\pmod{p^{n-k}}$.
\end{theorem}
\begin{proof}
    When $m=1$ is just Hensel's lemma. 
    \\

    If $m>1$, we can plug in $x_i=a_i$ for all $i\neq j$ and consider the polynomial $g(x_j)=f(a_1,\ldots,a_{j-1},x_j,a_{j+1},\ldots,a_m)$. It's clear that $g(a_j)\equiv 0\pmod{p^n}$ and $v_p(g'(a_j))=v_p\left(\frac{\partial f}{\partial x_j}(a)\right)=k$. So by Hensel's lemma, there exists $y_j\in \ZZ_p$ such that $g(y_j)=0$ and $y_j\equiv a_j \pmod{p^{n-k}}$. Thus, $y=(a_1,\ldots,a_{j-1},y_j,a_{j+1},\ldots,a_m)$ is the desired solution and $y\equiv a\pmod{p^{n-k}}$.
\end{proof}
\begin{corollary}
    Here are three useful special cases:
    \begin{enumerate}
        \item If $a\in (\ZZ_p)^m$ such that $f(a)\equiv 0\pmod p$ and there exists $1\leq j \leq m$ such that $\frac{\partial f}{\partial x_j}(a)$ is a unit in $\ZZ_p$, then there exists $y\in (\ZZ_p)^m$ such that $f(y)=0$ and $y\equiv a \pmod{p}$. This is the case when $n=1$ and $k=0$. 
        \item Suppose $p\neq 2$, let $f(x)=\sum_{i,j} a_{ij}x_ix_j$ with $a_{ij}=a_{ji}\in \ZZ_p$. If $\det(a_{ij})$ is a unit in $\ZZ_p$ and there exists a non-zero solution $r\in (\ZZ/p\ZZ)^n$ such that $f(r)\equiv t \pmod p$ for some $t\in \ZZ_p$, then there exists a non-trivial solution $y\in (\ZZ_p)^n$ such that $f(y)=t$ and $y\equiv r \pmod{p}$. 
        \\

        This is special case of (1). To prove it, we simply need to verify there exists $1\leq j \leq n$ such that $\frac{\partial f}{\partial x_j}(r)$ is a unit in $\ZZ_p$. Notice that \[\frac{\partial f}{\partial x_i}(r)=2\sum_{j}a_{ij}r_j\]

Let $U=(\ZZ_p)^{\times}$ be the group of units in $\ZZ_p$ and $U_n=1+p^n\ZZ_p$. $U_n$ form a decreasing sequence of subgroups of $U$, therefore, we have the following projective system:
\[\cdots \to \frac{U}{U_n}\to \frac{U}{U_{n-1}} \to \cdots \to \frac{U}{U_1}\to 0\]
The limit of this projective system is clearly $U$. 
\\

It's easy to check that $\frac{U}{U_n}\cong (\ZZ/p^n\ZZ)^{\times}$. In particular, $\frac{U}{U_1}\cong \FF_p^{\times}$. 
\begin{lemma}
    $\frac{U_1}{U_n}$ is finite of order $p^{n-1}$.
\end{lemma}
We can construct the canonical surjection from $\frac{U}{U_n} \surjto \frac{U}{U_1}$. The kernel is $\frac{U_1}{U_n}$. Therefore, we know that $|\frac{U_1}{U_n}|=\frac{|\frac{U}{U_n}|}{|\frac{U}{U_1}|}=\frac{p^{n-1}(p-1)}{p-1}=p^{n-1}$.

So the map is indeed a homomorphism. It's clearly surjective, and the kernel is $U_{n+1}$. So it's an isomorphism. So $|\frac{U_n}{U_{n+1}}|=p$.

\section{Chapter 4: Quadratic Forms over $\QQ_p$ and $\QQ$}
\subsection{Quadratic Forms over arbitrary fields $k$}
\subsubsection*{Definition}
We will assume the characteristic of $k$ is not 2 in this section and all vector spaces are finite dimensional over $k$.
\begin{definition}
    Let $V$ be a vector space over $k$. A quadratic form on $V$ is a function $Q: V\to k$ such that
    \begin{enumerate}
        \item $Q(ax)=a^2Q(x)$ for any $a\in k$ and $x\in V$.
        \item $(x,y)\mapsto Q(x+y)-Q(x)-Q(y)$ is a bilinear form. 
    \end{enumerate}
    We call pair $(V,Q)$ a quadratic space/module over $k$.
\end{definition}
\begin{definition}
    $x.y=\frac{1}{2}(Q(x+y)-Q(x)-Q(y))$ is called the bilinear form associated to $Q$. We can also write $Q(x)=x.x$. This establishes a one-to-one correspondence between quadratic forms and symmetric bilinear forms.
\end{definition}
\begin{definition}
    The morphism between quadratic spaces $(V_1,Q_1)$ and $(V_2,Q_2)$ is a linear map $f: V_1\to V_2$ such that $Q_2(f(x))=Q_1(x)$ for all $x\in V_1$.
\end{definition}
\begin{corollary}
    $f(x).f(y)=x.y$ for any morphism $f$ between quadratic spaces. 
\end{corollary}
Next, we will introduce the matrix representation of quadratic forms. Let $V$ be a vector space over $k$ with basis $\{e_1,\ldots,e_n\}$. For any $x=\sum_{i=1}^n x_ie_i$, we have
\[Q(x)=(\sum_{i=1}^n x_ie_i).(\sum_{j=1}^n x_je_j)=\sum_{i,j} a_{ij}x_ix_j\] where $a_{ij}=e_i.e_j$. Since $e_i.e_j=e_j.e_i$, we have $a_{ij}=a_{ji}$. So we can represent $Q$ by the symmetric matrix $M=(a_{ij})$. Furthermore, we can see that $Q(x)=x^TMx$. In general, $x.y=x^TM y$. 
\\

Notice that the matrix representation of a quadratic form depends on the choice of basis. If we change the basis by an invertible matrix $P$, then the new matrix representation will be $P^TMP$. So the $\det(M)$ is well-defined up to squares in $k$. We call $\det(M)$ the discriminant of $Q$ and denote it by $\disc(Q)$.

\subsubsection*{Orthgonality}
\begin{definition}
    $x$ and $y$ are called orthogonal if $x.y=0$. Let $H\subseteq V$ be a subset, we define $H^0$ be the set of all vectors in $V$ orthogonal to all vectors in $H$, and it's clearly a vector space. We call $H^0$ the orthogonal complement of $H$.
\end{definition}
\begin{definition}
    We define $\rad(V)$ to be $V^0$ when $V$ is the ambient space. 
\end{definition}
\begin{remark}
    If $U\subseteq V$, $\rad(U)$ is not the same as $U^0$ since $U^0$ is the set of vectors in $V$ orthogonal to all vectors in $U$, while $\rad(U)$ is the set of vectors in $U$ orthogonal to all vectors in $U$. In fact, $\rad(U)=U\cap U^0$.
\end{remark}
\begin{definition}
    $(V, Q)$ is called non-degenerate if $\rad(V)=0$.
\end{definition}
\begin{lemma}
    $(V, Q)$ is non-degenerate if and only if the matrix $M$ representing $Q$ is invertible if and only if $\disc(Q)\neq 0$.
\end{lemma}
\begin{proof}
    If $M$ is not invertible, then there exists a non-zero vector $x$ such that $Mx=0$. So $x.y=x^T M y=0$ for any $y\in V$, so $x\in \rad(V)$ and $(V,Q)$ is degenerate. Conversely, if $M$ is invertible, $Mx$ can be anything, so we can always find $x$ such that $z.x=z^T M x\neq 0$ for any non-zero $z$, so $\rad(V)=0$ and $(V,Q)$ is non-degenerate. 
\end{proof}
Let $U$ be a subsapce of $(V,Q)$. Let $U^*$ be the dual space of $U$. We can define a linear map $q_U: V\to U^*$ sending $x$ to the linear functional $y\mapsto x.y$. We can check that $\ker(q_U)=U^0$. In particular, if $U=V$, $\ker(q_V)=\rad(V)$. So if $(V,Q)$ is non-degenerate, $q_V$ is an isomorphism from $V$ to $V^*$.
\begin{definition}
    Let $U_1\cdots U_n$ be subspaces of $(V,Q)$. We say $V$ is the orthogonal direct sum of $U_1,\ldots,U_n$ if $V=\bigoplus_{i=1}^n U_i$ and $U_i\perp U_j$ for any $i\neq j$. We denote this by $V=U_1\po \cdots \po U_n$.
    \\

    In this case, we have $Q(x_1+\cdots +x_n)=Q(x_1)+\cdots +Q(x_n)$ for any $x_i\in U_i$.
\end{definition}
\begin{theorem}
    Let $V$ be a quadratic space over $k$ and $U$ be the complement of $\rad(V)$ in $V$, then $V=U\po\rad(V)$. Furthermore, $U$ is non-degenerate and $\rad(V)$ is isotropic (i.e. $Q(x)=0$ for any $x\in \rad(V)$).
\end{theorem}
\begin{proof}
    It's clear that $V=U\po\rad(V)$. Now, we show that $U$ is non-degenerate. For any $u\in \rad(U)\subseteq U\subseteq V$, $u.v=0$ for all $v\in U$. In fact, $u.(v+\rad(V))=0$ for all $v\in U$, that is, $u.v=0$ for all $v\in V$. So $u\in \rad(V)$. So $u=0$, implying $\rad(U)=0$ and $U$ is non-degenerate. Lastly, for any $x\in \rad(V)$, $Q(x)=x.x=0$, so $\rad(V)$ is isotropic.
\end{proof}
